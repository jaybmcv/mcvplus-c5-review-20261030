def load(name):
    return (original / name).read_text(encoding="utf-8-sig")

def save(name, value):
    (revision / name).write_text(value, encoding="utf-8", newline="\n")

spec = json.loads(load("SETUP-SPEC.json"))
# Replace exact reviewed installed-root bindings recursively, not arbitrary input paths.
def relocate(value):
    if isinstance(value, str):
        return value.replace(old_state, new_state)
    if isinstance(value, list):
        return [relocate(x) for x in value]
    if isinstance(value, dict):
        return {k: relocate(v) for k, v in value.items()}
    return value

}
for name in ("node", "nssm", "cloudflared"):
    spec["binary"][name]["source"] = f"inputs/binary/{name}.exe"
save("SETUP-SPEC.json", json.dumps(spec, indent=2) + "\n")

function Fault([string]$At){if($local -and ($LocalFault -eq $At -or ($LocalFault -eq 'ContainmentFailure' -and $At -eq 'AfterGrant'))){throw 'LOCAL_INJECTED_FAILURE'}}
function CheckPin($entry){Assert-C5Plain $entry.source;if((Get-Item -LiteralPath $entry.source).Length -ne $entry.bytes -or (Get-FileHash -LiteralPath $entry.source -Algorithm SHA256).Hash.ToLowerInvariant() -ne $entry.sha256){throw 'SOURCE_PIN_MISMATCH'}}
function Acl([string]$Path,[string[]]$Arguments){
 if($local){
  if(-not ([IO.Path]::GetFullPath($Path)).StartsWith($LocalRoot+'\',[StringComparison]::OrdinalIgnoreCase)){throw 'MOCK_ACL_OUTSIDE_ROOT'}
  if(-not(Test-Path -LiteralPath $Path)){throw 'MOCK_ACL_MISSING_PATH'}
  if($Arguments[0] -eq '/grant'){$pair=$Arguments[1] -split ':',2;$mockAcl[$Path+'|'+$pair[0]]=$pair[1]}
  elseif($Arguments[0] -eq '/remove:g'){$mockAcl.Remove($Path+'|'+$Arguments[1])}
  elseif($Arguments[0] -ne '/inheritance:r'){throw 'MOCK_ACL_UNSUPPORTED_OPERATION'}
  return
 }
 [void](Invoke-C5Native ($env:SystemRoot+'\System32\icacls.exe') (@($Path)+$Arguments) 'ACL' $bundle -TimeoutMs 10000)
}
function Grant([string]$Path,[string]$Identity,[string]$Permission){$grants.Add(@{path=$Path;identity=$Identity});Acl $Path @('/grant',($Identity+':'+$Permission))}
function CopyPinned($entry,[string]$dest){
 CheckPin $entry
 Assert-C5Plain $dest
 if(-not ([IO.Path]::GetFullPath($dest)).StartsWith($LocalRoot+'\',[StringComparison]::OrdinalIgnoreCase)){throw 'COPY_DESTINATION_OUTSIDE_LOCAL_ROOT'}
 New-Item -ItemType Directory -Path (Split-Path -Parent $dest) -Force | Out-Null
 [IO.File]::Copy($entry.source,$dest,$false)
 if((Get-Item -LiteralPath $dest).Length -ne $entry.bytes -or (Get-FileHash -LiteralPath $dest -Algorithm SHA256).Hash.ToLowerInvariant() -ne $entry.sha256){throw 'COPIED_HASH_MISMATCH'}
 if($script:firstCopy){$script:firstCopy=$false;Fault 'AfterFirstCopy'}
}
function ReadNewService($entry){if($local){if($mock.ContainsKey($entry.name)){return $mock[$entry.name]};return $null};return Get-C5Service $entry.name}
